{{- /* Email harvesting protection. - The address is never in the repository: it comes from the HUGO_CONTACT_EMAIL environment variable at build time (a GitHub Actions secret in production). - It is never in the HTML or the rendered DOM: profile links point to `#email`. The two halves (before/after the "@") are stored separately, each base64-encoded and reversed, so neither a source scan nor a base64-decoding scan finds an address. - It is only assembled when a person clicks or presses Enter on an email link (event.isTrusted), then handed straight to the mail app and the clipboard. Headless scrapers that render JavaScript still never see it in the page. */ -}} {{- $email := getenv "HUGO_CONTACT_EMAIL" | default site.Params.contact.email -}} {{- $parts := split (strings.TrimSpace (string $email)) "@" -}} {{- if eq (len $parts) 2 -}} {{- $enc := slice -}} {{- range $parts -}} {{- $rev := slice -}} {{- range split (base64Encode .) "" }}{{ $rev = slice . | append $rev }}{{ end -}} {{- $enc = $enc | append (delimit $rev "") -}} {{- end -}} {{- end -}}
